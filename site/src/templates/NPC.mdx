import Dropdown from '../components/Dropdown';
import EntityLink from '../components/EntityLink';
import Icon from '../components/Icon';
import InlineMeta from '../components/InlineMeta';
import MapSpot from '../components/MapSpot';

<div className="entity-header">
  {props.data.icon ? <div className="entity-header-icon"><Icon src={props.data.icon} alt={props.data.name} size={192} /></div> : null}
  <div className="entity-meta muted">
    {props.data.category && <span className="badge badge-category">{props.data.category}</span>}
    {props.data.inGame === false && <span className="badge badge-warn">cut from final build</span>}
    {props.data.locations.length > 0 && (
      <span className="badge badge-category">
        {props.data.locations.length} spawn{props.data.locations.length === 1 ? '' : 's'}
      </span>
    )}
  </div>
</div>

{props.data.comment.trim() && (
  <blockquote className="npc-comment">"{props.data.comment.trim()}"</blockquote>
)}

{props.data.startedMissions.length > 0 && (
  <Dropdown summary={`Gives ${props.data.startedMissions.length} mission${props.data.startedMissions.length === 1 ? '' : 's'}`} open>
    <ul className="ref-list">
      {props.data.startedMissions.map((m) => (
        <li key={m.id}><EntityLink entity={m} /></li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.journaledMissions.length > 0 && (
  <Dropdown summary={`Journals ${props.data.journaledMissions.length} mission${props.data.journaledMissions.length === 1 ? '' : 's'}`}>
    <ul className="ref-list">
      {props.data.journaledMissions.map((m) => (
        <li key={m.id}><EntityLink entity={m} /></li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.endedMissions.length > 0 && (
  <Dropdown summary={`Ends ${props.data.endedMissions.length} mission${props.data.endedMissions.length === 1 ? '' : 's'}`}>
    <ul className="ref-list">
      {props.data.endedMissions.map((m) => (
        <li key={m.id}><EntityLink entity={m} /></li>
      ))}
    </ul>
  </Dropdown>
)}


{props.data.transportRoutes.length > 0 && (
  <Dropdown summary={`Transportation (${props.data.transportRoutes.length})`} open>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>Route</th>
          <th>Start</th>
          <th>Landing</th>
        </tr>
      </thead>
      <tbody>
        {props.data.transportRoutes.map((route, i) => (
          <tr key={`${route.routeId}-${route.routeName}-${i}`}>
            <td>
              <strong>{route.routeName}</strong>
              {route.moveType && <div><InlineMeta>{route.moveType && <span>{route.moveType}</span>}</InlineMeta></div>}
            </td>
            <td>{route.start ? (
              <MapSpot x={route.start.x} y={route.start.y} z={route.start.z} areaId={route.start.areaId} title={route.start.areaZone} />
            ) : <span className="muted">—</span>}</td>
            <td>{route.landing ? (
              <MapSpot x={route.landing.x} y={route.landing.y} z={route.landing.z} areaId={route.landing.areaId} title={route.landing.areaZone} />
            ) : <span className="muted">—</span>}</td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

{props.data.vendorItems.length > 0 && (
  <Dropdown summary={`Sells ${props.data.vendorItems.length} item${props.data.vendorItems.length === 1 ? '' : 's'}`}>
    <ul className="reward-items">
      {props.data.vendorItems.map((v, i) => (
        <li key={`${v.ref.id}-${i}`}>
          <EntityLink entity={v.ref} />
          <InlineMeta leading>
            <span>{v.buyPrice.toLocaleString()} Taros</span>
            {v.rarity && <span>{v.rarity}</span>}
            {v.requiredLevel > 0 && <span>Lv {v.requiredLevel}</span>}
            {v.itemKind && <span>{v.itemKind}</span>}
          </InlineMeta>
        </li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.missionBarkers.length > 0 && (
  <Dropdown summary={`Mission barkers (${props.data.missionBarkers.length})`}>
    <ul className="barker-list">
      {props.data.missionBarkers.map((b, i) => (
        <li key={`${b.mission.id}-${i}`}>
          <EntityLink entity={b.mission} withIcon={false} />: <em>"{b.text}"</em>
        </li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.idleBarkers.length > 0 && (
  <Dropdown summary={`Idle barkers (${props.data.idleBarkers.length})`}>
    <ul className="barker-list">
      {props.data.idleBarkers.map((text, i) => (
        <li key={i}><em>"{text}"</em></li>
      ))}
    </ul>
  </Dropdown>
)}


{props.data.locations.length > 0 && (
  <Dropdown summary={`Spawn locations (${props.data.locations.length})`}>
    <ul className="ref-list location-list">
      {props.data.locations.map((loc, i) => (
        <li key={i}>
          <MapSpot
            x={loc.x}
            y={loc.y}
            z={loc.z}
            areaId={loc.areaId}
            title={loc.areaZone}
            instanceName={loc.instanceName}
            instanceID={loc.instanceID}
          />
        </li>
      ))}
    </ul>
  </Dropdown>
)}
