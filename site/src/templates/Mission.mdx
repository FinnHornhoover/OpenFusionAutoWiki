import Dropdown from '../components/Dropdown';
import EntityLink from '../components/EntityLink';
import RewardBox from '../components/RewardBox';
import MissionRequirementGraph from '../components/MissionRequirementGraph';
import TaskList from '../components/TaskList';

# {props.data.name}

<div className="entity-header">
  <div className="entity-meta muted">
    <span className="badge badge-category">Level {props.data.level}</span>
    {props.data.difficulty && <span className="badge badge-category">{props.data.difficulty}</span>}
    {props.data.type && <span className="badge badge-category">{props.data.type}</span>}
    {props.data.inGame === false && <span className="badge badge-warn">cut from final build</span>}
  </div>
</div>

<div className="mission-givers">
  {props.data.startNPC && (
    <div className="mission-giver-box">
      <span className="mission-giver-role">Start</span>
      <EntityLink entity={props.data.startNPC} iconSize={64} />
    </div>
  )}
  {props.data.journalNPC && props.data.journalNPC.id !== props.data.startNPC?.id && (
    <div className="mission-giver-box">
      <span className="mission-giver-role">Journal</span>
      <EntityLink entity={props.data.journalNPC} iconSize={64} />
    </div>
  )}
  {props.data.endNPC && props.data.endNPC.id !== props.data.startNPC?.id && (
    <div className="mission-giver-box">
      <span className="mission-giver-role">End</span>
      <EntityLink entity={props.data.endNPC} iconSize={64} />
    </div>
  )}
</div>

<MissionRequirementGraph mission={props.data} />

{(props.data.requiredMissions.length > 0 || props.data.requiredNano || props.data.requiredGuide) && (
  <Dropdown summary="Requirements">
    <ul className="ref-list">
      {props.data.requiredMissions.map((m) => (
        <li key={`mission-${m.id}`}><span className="task-label requirement-label">Mission:</span> <EntityLink entity={m} /></li>
      ))}
      {props.data.requiredNano && (
        <li key={`nano-${props.data.requiredNano.id}`}><span className="task-label requirement-label">Nano:</span> <EntityLink entity={props.data.requiredNano} /></li>
      )}
      {props.data.requiredGuide && (
        <li key="guide">
          <span className="task-label requirement-label">Guide:</span> {props.data.requiredGuideNpc ? <EntityLink entity={props.data.requiredGuideNpc} /> : props.data.requiredGuide}
        </li>
      )}
    </ul>
  </Dropdown>
)}

<Dropdown summary="Rewards" open>
  <RewardBox rewards={props.data.rewards} />
</Dropdown>

{props.data.requiredByMissions.length > 0 && (
  <Dropdown summary={`Required by ${props.data.requiredByMissions.length} mission${props.data.requiredByMissions.length === 1 ? '' : 's'}`}>
    <ul className="ref-list">
      {props.data.requiredByMissions.map((m) => (
        <li key={m.id}><EntityLink entity={m} /></li>
      ))}
    </ul>
  </Dropdown>
)}

## Tasks

<TaskList tasks={props.data.tasks} />

{props.data.barkers.length > 0 && (
  <Dropdown summary={`NPC barkers (${props.data.barkers.length})`}>
    <ul className="barker-list">
      {props.data.barkers.map((b, i) => (
        <li key={`${b.npc.id}-${i}`}>
          <EntityLink entity={b.npc} withIcon={false} />: <em>"{b.text}"</em>
        </li>
      ))}
    </ul>
  </Dropdown>
)}
