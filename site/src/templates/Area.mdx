import Dropdown from '../components/Dropdown';
import EntityLink from '../components/EntityLink';
import Icon from '../components/Icon';
import Minimap from '../components/Minimap';
import { formatDurationSeconds } from '../data/formatTime';

# {props.data.name}

<p className="entity-meta muted">
  {props.data.zoneName && <span className="badge badge-category">{props.data.zoneName}</span>}
  {props.data.infectedZone && <span className="badge badge-warn">has infected zone</span>}
  <span>{props.data.npcs.length} NPC types</span>
  <span>{props.data.mobs.length} monster types</span>
  {props.data.missionsStarting.length > 0 && (
    <span>{props.data.missionsStarting.length} mission{props.data.missionsStarting.length === 1 ? '' : 's'} start here</span>
  )}
</p>

{props.data.width > 0 && props.data.height > 0 && (
  <div className="area-map">
    <Minimap
      x={props.data.x + props.data.width / 2}
      y={props.data.y + props.data.height / 2}
      width={props.data.width}
      height={props.data.height}
      size={960}
      extent={Math.max(props.data.width, props.data.height) / 2 + 16384}
      title={props.data.fullName}
    />
  </div>
)}

{props.data.infectedZone && (
  <Dropdown summary="Infected Zone" open>
    <div className="infected-zone-card">
      <div className="infected-zone-header">
        {props.data.infectedZone.icon && <img src={props.data.infectedZone.icon} alt="" loading="lazy" className="icon infected-zone-icon" />}
        <div className="infected-zone-title">{props.data.infectedZone.name}</div>
      </div>
      <div className="infected-zone-details">
        {props.data.infectedZone.difficultyLabel && <div><strong>Difficulty:</strong> {props.data.infectedZone.difficultyLabel}</div>}
        {props.data.infectedZone.maxScore > 0 && <div><strong>Max score:</strong> {props.data.infectedZone.maxScore.toLocaleString()}</div>}
      </div>
      {props.data.infectedZone.description && <p className="infected-zone-description">{props.data.infectedZone.description}</p>}
    </div>
  </Dropdown>
)}

{props.data.missionsStarting.length > 0 && (
  <Dropdown summary={`Missions starting here (${props.data.missionsStarting.length})`} open={props.data.missionsStarting.length <= 20}>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>Mission</th>
        </tr>
      </thead>
      <tbody>
        {props.data.missionsStarting.map((m) => (
          <tr key={m.id}>
            <td><EntityLink entity={m} /></td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

{props.data.npcs.length > 0 && (
  <Dropdown summary={`NPCs (${props.data.npcs.length})`}>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>NPC</th>
          <th>Count</th>
        </tr>
      </thead>
      <tbody>
        {props.data.npcs.map((n) => (
          <tr key={n.ref.id}>
            <td><EntityLink entity={n.ref} /></td>
            <td>{n.instanceCount.toLocaleString()}</td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

{props.data.vendors.length > 0 && (
  <Dropdown summary={`Vendors (${props.data.vendors.length})`}>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>Vendor</th>
        </tr>
      </thead>
      <tbody>
        {props.data.vendors.map((v) => (
          <tr key={v.id}>
            <td><EntityLink entity={v} /></td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

{props.data.mobs.length > 0 && (
  <Dropdown summary={`Monsters (${props.data.mobs.length})`}>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>Monster</th>
          <th>Count</th>
          <th>Level</th>
          <th>HP</th>
        </tr>
      </thead>
      <tbody>
        {props.data.mobs.map((m) => (
          <tr key={m.ref.id}>
            <td><EntityLink entity={m.ref} /></td>
            <td>{m.instanceCount.toLocaleString()}</td>
            <td>{m.level > 0 ? m.level : <span className="muted">—</span>}</td>
            <td>{m.hp > 0 ? m.hp.toLocaleString() : <span className="muted">—</span>}</td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

{props.data.transportation.length > 0 && (
  <Dropdown summary={`Transportation (${props.data.transportation.length})`}>
    <ul className="ref-list">
      {props.data.transportation.map((t, i) => (
        <li key={`${t.routeId}-${i}`}>
          <strong>{t.routeName}</strong>
          {t.moveType && <span className="muted"> · {t.moveType}</span>}
          {t.startNpc && <span className="muted"> · operated by <EntityLink entity={t.startNpc} withIcon={false} /></span>}
        </li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.instanceWarps.length > 0 && (
  <Dropdown summary={`Instance warps (${props.data.instanceWarps.length})`}>
    <ul className="ref-list">
      {props.data.instanceWarps.map((w, i) => (
        <li key={`${w.instanceID}-${i}`}>
          <strong>→ {w.instanceName}</strong>
          {w.npc && <span className="muted"> · via <EntityLink entity={w.npc} withIcon={false} /></span>}
          {w.requiredMinLevel > 0 && <span className="muted"> · Lv {w.requiredMinLevel}+</span>}
          {w.requiredItem && <span className="muted"> · needs <EntityLink entity={w.requiredItem} withIcon={false} /></span>}
        </li>
      ))}
    </ul>
  </Dropdown>
)}

{props.data.eggs.length > 0 && (
  <Dropdown summary={`Eggs (${props.data.eggs.length})`}>
    <table className="location-table source-table">
      <thead>
        <tr>
          <th>Egg</th>
          <th>Type</th>
          <th>Contains</th>
          <th>Power</th>
          <th>Duration</th>
        </tr>
      </thead>
      <tbody>
        {props.data.eggs.map((e, i) => (
          <tr key={i}>
            <td>{e.typeName || <span className="muted">Egg</span>}</td>
            <td>{e.typeComment || <span className="muted">—</span>}</td>
            <td>{e.crateItem ? <EntityLink entity={e.crateItem} /> : <span className="muted">—</span>}</td>
            <td>
              {e.effectName ? (
                <>
                  {e.effectIcon && <Icon src={e.effectIcon} alt="" size={32} />}
                  {' '}{e.effectName}
                </>
              ) : <span className="muted">—</span>}
            </td>
            <td>{e.effectDuration > 0 ? formatDurationSeconds(e.effectDuration) : <span className="muted">—</span>}</td>
          </tr>
        ))}
      </tbody>
    </table>
  </Dropdown>
)}

<p className="muted alias-ids">Area key: <code>{props.data.fullName}</code></p>
